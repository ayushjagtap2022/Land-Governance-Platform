## BREAKPOINT 1: API Client & Auth Store

### 1a. Install axios
```bash
cd apps/frontend && npm install axios zustand
```

### 1b. Create `src/lib/api.ts`
```typescript
// Base axios instance with interceptors
import axios from 'axios';

const api = axios.create({
  baseURL: import.meta.env.VITE_API_URL || 'http://127.0.0.1:8000/api/v1',
  headers: { 'Content-Type': 'application/json' },
});

// Request interceptor: attach JWT token
api.interceptors.request.use((config) => {
  const token = localStorage.getItem('access_token');
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

// Response interceptor: catch 401 and force logout
api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem('access_token');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

export default api;
```

### 1c. Create `src/stores/authStore.ts` (Zustand)
```typescript
// Global auth state
import { create } from 'zustand';

type User = {
  id: string;
  email: string;
  full_name: string;
  role: 'public' | 'researcher' | 'official' | 'institution' | 'super_admin';
  institution?: string;
  is_active: boolean;
  is_verified: boolean;
  created_at: string;
};

type AuthState = {
  user: User | null;
  token: string | null;
  isAuthenticated: boolean;
  login: (token: string, user: User) => void;
  logout: () => void;
  hydrate: () => void; // Restore from localStorage on app load
};

export const useAuthStore = create<AuthState>((set) => ({
  user: null,
  token: null,
  isAuthenticated: false,
  login: (token, user) => {
    localStorage.setItem('access_token', token);
    localStorage.setItem('user', JSON.stringify(user));
    set({ user, token, isAuthenticated: true });
  },
  logout: () => {
    localStorage.removeItem('access_token');
    localStorage.removeItem('user');
    set({ user: null, token: null, isAuthenticated: false });
  },
  hydrate: () => {
    const token = localStorage.getItem('access_token');
    const userStr = localStorage.getItem('user');
    if (token && userStr) {
      set({ token, user: JSON.parse(userStr), isAuthenticated: true });
    }
  },
}));
```

## BREAKPOINT 5: WebSocket Chat (Workspaces)

### File: `src/pages/workspaces-page.tsx`
When a user opens a workspace detail view (line 84), add a chat panel.

Create a custom hook `src/hooks/use-chat.ts`:
```typescript
export function useChat(workspaceId: string) {
  const token = localStorage.getItem('access_token');
  const [messages, setMessages] = useState<Message[]>([]);
  const wsRef = useRef<WebSocket | null>(null);

  useEffect(() => {
    const ws = new WebSocket(
      `ws://127.0.0.1:8000/api/v1/ws/chat/${workspaceId}?token=${token}`
    );
    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      setMessages(prev => [...prev, data]);
    };
    wsRef.current = ws;
    return () => ws.close();
  }, [workspaceId, token]);

  const sendMessage = (content: string) => {
    wsRef.current?.send(JSON.stringify({ content }));
  };

  return { messages, sendMessage };
}
```

Add automatic reconnection with exponential backoff if the connection drops.

## BREAKPOINT 6: Real-Time Notifications

Create `src/hooks/use-notifications.ts`:
```typescript
export function useNotifications() {
  const token = localStorage.getItem('access_token');
  
  useEffect(() => {
    if (!token) return;
    const ws = new WebSocket(
      `ws://127.0.0.1:8000/api/v1/notifications/ws/notifications?token=${token}`
    );
    ws.onmessage = (event) => {
      const { data } = JSON.parse(event.data);
      // Use sonner to show a toast
      toast(data.title, { description: data.content });
    };
    return () => ws.close();
  }, [token]);
}
```

Call `useNotifications()` inside the `Shell` component in `App.tsx` so it runs globally whenever a user is logged in. This means that if a Super Admin approves a grant proposal on one screen, the researcher instantly gets a beautiful toast popup on their screen.
